Shared UI Component Guidelines

Design specifications, component guidelines, and HTML boilerplate. Use the ght- class namespace prefix.

1. How to Setup & Reset

To run the templates and render styles correctly, ensure your base HTML files load these assets in the document header:

<!-- Fonts -->
<link href="https://fonts.googleapis.com/css2?family=DM+Sans:wght@400;500;700;900&display=swap" rel="stylesheet">

<!-- Iconify -->
<script src="https://code.iconify.design/iconify-icon/2.1.0/iconify-icon.min.js"></script>

<!-- Tailwind CSS with Project Theme Config -->
<script src="https://cdn.tailwindcss.com"></script>
<script>
  tailwind.config = {
    theme: {
      extend: {
        colors: {
          'ght-primary': '#8dc63f',
          'ght-primary-hover': '#7cb342',
          'ght-accent': '#f7931e',
          'ght-dark': '#0A0A0A',
          'ght-muted': 'rgba(0, 0, 0, 0.45)',
          'ght-border': 'rgba(0, 0, 0, 0.08)',
          'ght-background': '#FAFAFA',
          'ght-surface': '#F4F4F4'
        },
        fontFamily: {
          sans: ['"DM Sans"', 'sans-serif'],
        }
      }
    }
  }
</script>
